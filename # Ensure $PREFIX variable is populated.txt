# Ensure $PREFIX variable is populated
if (-not $PREFIX) {$PREFIX = "yt-pipeline-akshath" }

# Fetch AWS Account ID dynamically
$ACCOUNT_ID = (aws sts get-caller-identity --query "Account" --output text)
$ROLE_ARN = "arn:aws:iam::${ACCOUNT_ID}:role/yt-pipeline-glue-role"

# 1. Create Glue Data Catalog Databases
aws glue create-database --database-input '{"Name":"yt_pipeline_bronze_db","Description":"Bronze raw data layer"}'
aws glue create-database --database-input '{"Name":"yt_pipeline_silver_db","Description":"Silver cleaned data layer"}'
aws glue create-database --database-input '{"Name":"yt_pipeline_gold_db","Description":"Gold aggregated analytics layer"}'

# 2. Build target configuration JSON for Glue Crawler
$targets = @{
    S3Targets = @(
        @{ Path = "s3://$PREFIX-bronze-dev/youtube/raw_statistics/" },
        @{ Path = "s3://$PREFIX-bronze-dev/youtube/reference_data/" }
    )
} | ConvertTo-Json -Depth 3
$targets | Out-File -Encoding ascii crawler-targets.json

# 3. Create the Glue Crawler
aws glue create-crawler `
    --name "yt-pipeline-bronze-crawler" `
    --role $ROLE_ARN `
    --database-name "yt_pipeline_bronze_db" `
    --targets file://crawler-targets.json

# Clean up temp file
Remove-Item crawler-targets.json

# 4. Start the Crawler
aws glue start-crawler --name "yt-pipeline-bronze-crawler"
